cmake_minimum_required(VERSION 3.16)
project(serial_frame VERSION 0.3.0 LANGUAGES C)

add_library(serial_frame c/src/serial_frame.c)
target_include_directories(serial_frame PUBLIC c/include)
set_target_properties(serial_frame PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(serial_frame PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(test_serial_frame c/tests/test_serial_frame.c)
    target_link_libraries(test_serial_frame PRIVATE serial_frame)
    set_target_properties(test_serial_frame PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(test_serial_frame PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME serial_frame_core COMMAND test_serial_frame)
    add_executable(sf_vector_tool c/tests/vector_tool.c)
    target_link_libraries(sf_vector_tool PRIVATE serial_frame)
    set_target_properties(sf_vector_tool PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
endif()
